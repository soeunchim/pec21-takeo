function onOpen() {
  const ui = SpreadsheetApp.getUi();
  ui.createMenu('គ.ក.ប ២០២៧')
    .addItem('⚡ រៀបចំតារាងទិន្នន័យស្វ័យប្រវត្តិ (Initialize Sheets)', 'initializeAllSheets')
    .addItem('📊 ពិនិត្យស្ថិតិបេក្ខជនសរុប (Check Summary)', 'showSummaryAlert')
    .addToUi();
}

function doGet(e) {
  try {
    const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'get_all';
    
    if (action === 'ping') {
      return responseJSON({ status: 'success', message: 'Google Sheet API is alive', time: new Date().toISOString() });
    }
    
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const result = {
      settings: readSheetObjects(ss, 'Settings'),
      districts: readSheetObjects(ss, 'Districts'),
      communes: readSheetObjects(ss, 'Communes'),
      users: readSheetObjects(ss, 'Users'),
      pollingStations: readSheetObjects(ss, 'PollingStations'),
      candidates: readSheetObjects(ss, 'Candidates'),
      lastUpdated: new Date().toISOString()
    };
    
    return responseJSON({ status: 'success', data: result });
  } catch (err) {
    return responseJSON({ status: 'error', message: err.toString() });
  }
}

function doPost(e) {
  try {
    let payload = {};
    if (e && e.postData && e.postData.contents) {
      payload = JSON.parse(e.postData.contents);
    }
    
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const action = payload.action || 'sync_all';
    
    if (action === 'sync_all' && payload.data) {
      const data = payload.data;
      if (data.settings) writeSheetObjects(ss, 'Settings', Array.isArray(data.settings) ? data.settings : [data.settings]);
      if (data.users && data.users.length) writeSheetObjects(ss, 'Users', data.users);
      if (data.districts && data.districts.length) writeSheetObjects(ss, 'Districts', data.districts);
      if (data.communes && data.communes.length) writeSheetObjects(ss, 'Communes', data.communes);
      if (data.pollingStations && data.pollingStations.length) writeSheetObjects(ss, 'PollingStations', data.pollingStations);
      if (data.candidates && data.candidates.length) writeSheetObjects(ss, 'Candidates', data.candidates);
      
      logAudit(ss, 'SYNC_ALL', payload.user || 'system', `Updated ${data.candidates ? data.candidates.length : 0} candidates`);
      
      return responseJSON({
        status: 'success',
        message: 'Synchronized all data to Google Sheets successfully',
        timestamp: new Date().toISOString()
      });
    }
    
    if (action === 'save_candidate' && payload.candidate) {
      upsertSingleCandidate(ss, payload.candidate);
      return responseJSON({ status: 'success', message: 'Candidate saved' });
    }
    
    return responseJSON({ status: 'error', message: 'Unknown action: ' + action });
  } catch (err) {
    return responseJSON({ status: 'error', message: err.toString() });
  }
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function readSheetObjects(ss, sheetName) {
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];
  
  const headers = data[0];
  const list = [];
  for (let r = 1; r < data.length; r++) {
    const row = data[r];
    const obj = {};
    let hasData = false;
    for (let c = 0; c < headers.length; c++) {
      const h = headers[c];
      if (h) {
        let val = row[c];
        if (typeof val === 'string' && (val.startsWith('{') || val.startsWith('['))) {
          try { val = JSON.parse(val); } catch(e){}
        }
        obj[h] = val;
        if (val !== '' && val !== null && val !== undefined) hasData = true;
      }
    }
    if (hasData) list.push(obj);
  }
  return list;
}

function writeSheetObjects(ss, sheetName, objects) {
  if (!objects || !objects.length) return;
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
  }
  
  // Extract all unique keys
  const keySet = new Set();
  objects.forEach(o => {
    Object.keys(o).forEach(k => keySet.add(k));
  });
  const headers = Array.from(keySet);
  
  const rows = [headers];
  objects.forEach(o => {
    const row = headers.map(h => {
      const val = o[h];
      if (val === undefined || val === null) return '';
      if (typeof val === 'object') return JSON.stringify(val);
      return val;
    });
    rows.push(row);
  });
  
  sheet.clear();
  sheet.getRange(1, 1, rows.length, headers.length).setValues(rows);
  
  // Format header row
  const headerRange = sheet.getRange(1, 1, 1, headers.length);
  headerRange.setBackground('#1a365d').setFontColor('#ffffff').setFontWeight('bold');
  sheet.setFrozenRows(1);
}

function upsertSingleCandidate(ss, cand) {
  let sheet = ss.getSheetByName('Candidates');
  if (!sheet) {
    writeSheetObjects(ss, 'Candidates', [cand]);
    return;
  }
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const idCol = headers.indexOf('id');
  if (idCol === -1) {
    writeSheetObjects(ss, 'Candidates', [cand]);
    return;
  }
  
  let targetRow = -1;
  for (let r = 1; r < data.length; r++) {
    if (String(data[r][idCol]) === String(cand.id)) {
      targetRow = r + 1;
      break;
    }
  }
  
  const rowVals = headers.map(h => {
    const v = cand[h];
    if (v === undefined || v === null) return '';
    if (typeof v === 'object') return JSON.stringify(v);
    return v;
  });
  
  if (targetRow > 0) {
    sheet.getRange(targetRow, 1, 1, headers.length).setValues([rowVals]);
  } else {
    sheet.appendRow(rowVals);
  }
}

function logAudit(ss, action, user, detail) {
  let sheet = ss.getSheetByName('AuditLogs');
  if (!sheet) {
    sheet = ss.insertSheet('AuditLogs');
    sheet.appendRow(['Timestamp', 'Action', 'User', 'Detail']);
    sheet.getRange(1, 1, 1, 4).setBackground('#2d3748').setFontColor('#fff').setFontWeight('bold');
  }
  sheet.appendRow([new Date().toISOString(), action, user, detail]);
}

function initializeAllSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ['Settings', 'Users', 'Districts', 'Communes', 'PollingStations', 'Candidates', 'AuditLogs'].forEach(name => {
    if (!ss.getSheetByName(name)) ss.insertSheet(name);
  });
  SpreadsheetApp.getUi().alert('ជោគជ័យ', 'បានបង្កើត Tabs ទិន្នន័យទាំងអស់រួចរាល់ហើយ!', SpreadsheetApp.getUi().ButtonSet.OK);
}

function showSummaryAlert() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const candSheet = ss.getSheetByName('Candidates');
  const count = candSheet ? Math.max(0, candSheet.getLastRow() - 1) : 0;
  SpreadsheetApp.getUi().alert('ស្ថិតិ គ.ក.ប ២០២៧', 'ចំនួនបេក្ខជនសរុបក្នុងប្រព័ន្ធ៖ ' + count + ' នាក់', SpreadsheetApp.getUi().ButtonSet.OK);
}
